// 1. 피셔-예이츠 무작위 셔플 알고리즘
function shuffle(array) {
  let currentIndex = array.length, randomIndex;
  while (currentIndex !== 0) {
    randomIndex = Math.floor(Math.random() * currentIndex);
    currentIndex--;
    [array[currentIndex], array[randomIndex]] = [array[randomIndex], array[currentIndex]];
  }
  return array;
}

// 2. 입력 데이터 가져오기
function getStudents() {
  const textarea = document.getElementById('studentInput');
  return textarea.value.split('\n').map(name => name.trim()).filter(name => name !== "");
}

// 3. 그리드 및 카드 동적 렌더링
function renderGrid(shuffledStudents = []) {
  const rows = parseInt(document.getElementById('rows').value);
  const cols = parseInt(document.getElementById('cols').value);
  const gridContainer = document.getElementById('seatingGrid');
  
  // Tailwind Grid Template Columns 동적 적용
  gridContainer.style.gridTemplateColumns = `repeat(${cols}, minmax(100px, 1fr))`;
  gridContainer.innerHTML = '';

  const totalSeats = rows * cols;

  for (let i = 0; i < totalSeats; i++) {
    const studentName = shuffledStudents[i] || "❌ 빈자리";
    
    const seatDiv = document.createElement('div');
    seatDiv.className = "bg-indigo-50 border-2 border-indigo-200 rounded-xl p-4 text-center font-bold text-indigo-900 shadow-sm flex items-center justify-center min-h-[70px] cursor-pointer hover:scale-105 transition-transform duration-200";
    seatDiv.innerText = studentName;
    
    // 드래그 앤 드롭 속성 부여 (추후 확장용)
    seatDiv.draggable = true;
    
    gridContainer.appendChild(seatDiv);
  }
}

// 4. 랜덤 배치 실행 함수
function triggerRandomShuffle() {
  const students = getStudents();
  if (students.length === 0) {
    alert("학생 이름을 입력해주세요!");
    return;
  }
  
  // 애니메이션 연출을 위한 타이머 처리 (셔플하는 느낌 주기)
  let count = 0;
  const interval = setInterval(() => {
    const temporaryShuffle = shuffle([...students]);
    renderGrid(temporaryShuffle);
    count++;
    if (count > 8) {
      clearInterval(interval);
      const finalResult = shuffle([...students]);
      renderGrid(finalResult);
    }
  }, 100);
}

// 초기 화면 빌드
window.onload = () => {
  renderGrid();
};
